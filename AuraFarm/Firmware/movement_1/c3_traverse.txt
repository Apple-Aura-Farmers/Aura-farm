// Phase 1b: X/Y traverse endurance test (X GPIO3, Y GPIO4), ESP32-C3 Super Mini.
// Board: ESP32-C3 Super Mini, Arduino-ESP32 core 3.x, USB CDC On Boot: Enabled.
// C3 runs from the Mac's USB. Servos run from the HW-131 rail (separate power, shared ground).
// PWM: built-in LEDC API (ledcAttach / ledcWrite / ledcDetach). No Bluetooth, no Wi-Fi.
// One traversal: X -> min -> max, then Y -> min -> max, then X -> home, then Y -> home.
// Only one servo ever has a signal: each move finishes, settles, switches off and rests
// before the next move starts. Limits and home come from the calibration saved in flash;
// anything not saved uses the defaults below (full 0-180 range, home 90).

#include <Preferences.h>
#include "esp_system.h"

// ---------------- Settings ----------------
const int  NUM_SERVOS = 2;
const int  SERVO_PINS[NUM_SERVOS]  = {3, 4};
const char SERVO_NAMES[NUM_SERVOS] = {'X', 'Y'};

const int DEFAULT_MIN  = 0;     // used when no calibration is saved
const int DEFAULT_MAX  = 180;   // (0-180 checked clean on these servos with 'sweep')
const int DEFAULT_HOME = 90;

const int STEP_DEG     = 2;     // move 2 deg ...
const int STEP_MS      = 20;    // ... every 20 ms
const int SETTLE_MS    = 300;   // hold after reaching target, before signal off
const int GAP_MS       = 500;   // rest with signal off after every move (also between servos)
const int BOOT_WAIT_MS = 500;   // pause after boot before first attach
const int CYCLES       = 100;   // traversals

const int PULSE_MIN_US = 500;   // pulse width at 0 deg (typical SG90)
const int PULSE_MAX_US = 2400;  // pulse width at 180 deg

const int PWM_FREQ_HZ = 50;
const int PWM_BITS    = 14;

const unsigned long SERIAL_WAIT_MS = 5000;  // wait this long for Serial Monitor at boot
const int LOG_SIZE = 10;                    // boots kept in the saved boot log

// ---------------- State ----------------
int  minAngle[NUM_SERVOS];
int  maxAngle[NUM_SERVOS];
int  homeAngle[NUM_SERVOS];
int  currentAngle[NUM_SERVOS];   // last commanded angle per servo
int  cycle = 0;
bool done = false;
int  thisReason = 0;

struct BootLog {
  uint32_t boots;
  uint8_t  reason[LOG_SIZE];
  uint16_t cycle[LOG_SIZE];
};
BootLog bootLog;
Preferences logPrefs;

// ---------------- Reset reason ----------------
const char* resetName(int r) {
  switch (r) {
    case ESP_RST_POWERON:   return "ESP_RST_POWERON";
    case ESP_RST_BROWNOUT:  return "ESP_RST_BROWNOUT";
    case ESP_RST_EXT:       return "ESP_RST_EXT";
    case ESP_RST_SW:        return "ESP_RST_SW";
    case ESP_RST_PANIC:     return "ESP_RST_PANIC";
    case ESP_RST_INT_WDT:   return "ESP_RST_INT_WDT";
    case ESP_RST_TASK_WDT:  return "ESP_RST_TASK_WDT";
    case ESP_RST_WDT:       return "ESP_RST_WDT";
    case ESP_RST_DEEPSLEEP: return "ESP_RST_DEEPSLEEP";
    case ESP_RST_SDIO:      return "ESP_RST_SDIO";
    case ESP_RST_USB:       return "ESP_RST_USB";
    case ESP_RST_JTAG:      return "ESP_RST_JTAG";
    case ESP_RST_UNKNOWN:   return "ESP_RST_UNKNOWN";
    default:                return "OTHER";
  }
}

// ---------------- Helpers ----------------
void stamp() {
  Serial.printf("[%9.3f s] ", millis() / 1000.0);
}

uint32_t angleToDuty(int angle) {
  uint32_t us = PULSE_MIN_US + (uint32_t)(PULSE_MAX_US - PULSE_MIN_US) * angle / 180;
  uint32_t periodUs = 1000000UL / PWM_FREQ_HZ;
  return us * ((1UL << PWM_BITS) - 1) / periodUs;
}

void printServo(int i) {
  Serial.printf("servo %c (GPIO%d)", SERVO_NAMES[i], SERVO_PINS[i]);
}

// ---------------- Limits and home (from calibration) ----------------
// Uses the calibration saved in flash; any value not saved (or not valid) falls back to the
// defaults for that servo.
void loadLimits() {
  Preferences cal;
  cal.begin("calib", true);  // read-only
  const char* minKeys[NUM_SERVOS]  = {"xmin", "ymin"};
  const char* maxKeys[NUM_SERVOS]  = {"xmax", "ymax"};
  const char* homeKeys[NUM_SERVOS] = {"xhome", "yhome"};
  for (int i = 0; i < NUM_SERVOS; i++) {
    int mn = cal.getInt(minKeys[i], -1);
    int mx = cal.getInt(maxKeys[i], -1);
    int hm = cal.getInt(homeKeys[i], -1);
    bool limitsOk = mn >= DEFAULT_MIN && mx <= DEFAULT_MAX && mn < mx;
    minAngle[i] = limitsOk ? mn : DEFAULT_MIN;
    maxAngle[i] = limitsOk ? mx : DEFAULT_MAX;
    bool homeOk = hm >= minAngle[i] && hm <= maxAngle[i];
    homeAngle[i] = homeOk ? hm : DEFAULT_HOME;
    printServo(i);
    Serial.printf(" | min %d, max %d (%s) | home %d (%s)\n",
                  minAngle[i], maxAngle[i], limitsOk ? "saved" : "default",
                  homeAngle[i], homeOk ? "saved" : "default");
  }
  cal.end();
}

// ---------------- Saved boot log (flash) ----------------
void saveLog() {
  logPrefs.putBytes("log", &bootLog, sizeof(bootLog));
}

void printLog() {
  Serial.println("---- saved boot log (oldest first) ----");
  uint32_t first = bootLog.boots > LOG_SIZE ? bootLog.boots - LOG_SIZE + 1 : 1;
  for (uint32_t b = first; b <= bootLog.boots; b++) {
    int i = b % LOG_SIZE;
    Serial.printf("boot %lu | reset reason: %s | reached cycle %u/%d\n",
                  (unsigned long)b, resetName(bootLog.reason[i]), bootLog.cycle[i], CYCLES);
  }
  Serial.println("(send any key to reprint, 'c' to clear)");
  Serial.println("---------------------------------------");
}

// Called between traversals: any key reprints the boot log, 'c' clears it first.
void checkSerial() {
  bool any = false, clear = false;
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'c' || c == 'C') clear = true;
    if (c != '\n' && c != '\r') any = true;
  }
  if (!any) return;
  if (clear) {
    memset(&bootLog, 0, sizeof(bootLog));
    bootLog.boots = 1;  // keep only the current boot
    bootLog.reason[1] = thisReason;
    bootLog.cycle[1] = cycle;
    saveLog();
    Serial.println("Boot log cleared.");
  }
  printLog();
}

// ---------------- Servo ----------------
// Start the signal at this servo's last commanded angle so it does not twitch.
void servoOn(int i) {
  if (!ledcAttach(SERVO_PINS[i], PWM_FREQ_HZ, PWM_BITS)) {
    Serial.println("ERROR: ledcAttach failed");
  }
  ledcWrite(SERVO_PINS[i], angleToDuty(currentAngle[i]));
}

// Stop the signal and hold the line low (no pulses = servo idle).
void servoOff(int i) {
  ledcDetach(SERVO_PINS[i]);
  pinMode(SERVO_PINS[i], OUTPUT);
  digitalWrite(SERVO_PINS[i], LOW);
}

void logMove(int i, const char* label, int target, const char* phase) {
  stamp();
  Serial.printf("TR cycle %d/%d | ", cycle, CYCLES);
  printServo(i);
  Serial.printf(" | -> %d deg (%s) | %s\n", target, label, phase);
}

// One full move of one servo: signal on, small steps to target, settle, signal off, rest.
void moveTo(int i, int target, const char* label) {
  target = constrain(target, minAngle[i], maxAngle[i]);
  logMove(i, label, target, "start");
  servoOn(i);
  while (currentAngle[i] != target) {
    delay(STEP_MS);
    currentAngle[i] += constrain(target - currentAngle[i], -STEP_DEG, STEP_DEG);
    ledcWrite(SERVO_PINS[i], angleToDuty(currentAngle[i]));
  }
  delay(SETTLE_MS);
  servoOff(i);
  logMove(i, label, target, "end");
  delay(GAP_MS);
}

// ---------------- Main ----------------
void setup() {
  for (int i = 0; i < NUM_SERVOS; i++) {
    pinMode(SERVO_PINS[i], OUTPUT);
    digitalWrite(SERVO_PINS[i], LOW);
  }

  Serial.begin(115200);
  unsigned long t0 = millis();
  while (!Serial && millis() - t0 < SERIAL_WAIT_MS) delay(10);

  thisReason = esp_reset_reason();

  // Record this boot in the saved boot log.
  logPrefs.begin("travlog", false);
  if (logPrefs.getBytes("log", &bootLog, sizeof(bootLog)) != sizeof(bootLog)) {
    memset(&bootLog, 0, sizeof(bootLog));
  }
  bootLog.boots++;
  bootLog.reason[bootLog.boots % LOG_SIZE] = thisReason;
  bootLog.cycle[bootLog.boots % LOG_SIZE] = 0;
  saveLog();

  Serial.println();
  Serial.println("=== Phase 1b: X/Y traverse test (ESP32-C3 Super Mini) ===");
  Serial.printf("Reset reason: %s (%d)\n", resetName(thisReason), thisReason);
  Serial.printf("Servos X (GPIO3), Y (GPIO4) | step %d deg / %d ms | settle %d ms | gap %d ms | traversals %d\n",
                STEP_DEG, STEP_MS, SETTLE_MS, GAP_MS, CYCLES);
  loadLimits();
  printLog();

  // Attach X, then Y, each with one jump to its home (position unknown after power-up).
  delay(BOOT_WAIT_MS);
  for (int i = 0; i < NUM_SERVOS; i++) {
    stamp();
    Serial.print("startup | ");
    printServo(i);
    Serial.printf(" | attach at home %d deg (one direct jump, position unknown)\n", homeAngle[i]);
    currentAngle[i] = homeAngle[i];
    servoOn(i);
    delay(SETTLE_MS);
    servoOff(i);
    delay(GAP_MS);
  }
}

void loop() {
  checkSerial();
  if (done) {
    delay(100);
    return;
  }

  cycle++;
  moveTo(0, minAngle[0], "min");    // full X sweep
  moveTo(0, maxAngle[0], "max");
  moveTo(1, minAngle[1], "min");    // full Y sweep
  moveTo(1, maxAngle[1], "max");
  moveTo(0, homeAngle[0], "home");  // back home, X then Y
  moveTo(1, homeAngle[1], "home");

  bootLog.cycle[bootLog.boots % LOG_SIZE] = cycle;
  saveLog();

  if (cycle >= CYCLES) {
    done = true;
    stamp();
    Serial.printf("TRAVERSE DONE: %d traversals, both servos at home, no restart during this boot\n", CYCLES);
  }
}
